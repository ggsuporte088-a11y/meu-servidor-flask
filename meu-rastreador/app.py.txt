import os
from flask import Flask, request
import requests

app = Flask(__name__)

@app.route('/')
def home():
    # Captura o IP real do visitante através do proxy do Render
    ip = request.headers.get('X-Forwarded-For', request.remote_addr)
    if ip and ',' in ip:
        ip = ip.split(',')[0].strip()

    # Consulta a API de localização usando o IP capturado
    try:
        resposta = requests.get(f'http://ip-api.com/json/{ip}', timeout=5).json()
        cidade = resposta.get('city', 'Desconhecida')
        pais = resposta.get('country', 'Desconhecido')
        provedor = resposta.get('isp', 'Desconhecido')
    except Exception:
        cidade, pais, provedor = "Erro", "Erro", "Erro"

    # Página simples que o visitante vai ver ao clicar
    return f"""
    <!DOCTYPE html>
    <html>
    <head>
        <title>Página de Boas-Vindas</title>
        <style>
            body {{ font-family: Arial, sans-serif; text-align: center; margin-top: 50px; background-color: #f4f4f9; }}
            .card {{ background: white; padding: 30px; display: inline-block; border-radius: 10px; box-shadow: 0 0 10px rgba(0,0,0,0.1); }}
            h1 {{ color: #333; }}
            p {{ font-size: 18px; color: #666; }}
        </style>
    </head>
    <body>
        <div class="card">
            <h1>Conexão Estabelecida com Sucesso!</h1>
            <p>Seu IP detectado: <strong>{ip}</strong></p>
            <p>Localização aproximada: <strong>{cidade}, {pais}</strong></p>
            <p>Provedor: <strong>{provedor}</strong></p>
        </div>
    </body>
    </html>
    """

if __name__ == '__main__':
    port = int(os.environ.get('PORT', 5000))
    app.run(host='0.0.0.0', port=port)
